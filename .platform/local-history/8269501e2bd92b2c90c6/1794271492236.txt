// === Файл: UsageStatsHelper.kt ===
package com.example.apptime

import android.app.AppOpsManager
import android.app.usage.UsageEvents
import android.app.usage.UsageStatsManager
import android.content.Context
import android.content.pm.ApplicationInfo
import android.content.pm.PackageManager
import android.graphics.drawable.Drawable
import android.os.Build
import android.os.Process
import java.util.Calendar
import java.util.concurrent.ConcurrentHashMap
import kotlin.math.max
import kotlin.math.min

class UsageStatsHelper(private val context: Context) {

    private val usageStatsManager =
        context.getSystemService(Context.USAGE_STATS_SERVICE) as UsageStatsManager
    private val packageManager: PackageManager = context.packageManager

    private val appNameCache = ConcurrentHashMap<String, String>()
    private val appIconCache = ConcurrentHashMap<String, Drawable>()
    private val systemFlagCache = ConcurrentHashMap<String, Boolean>()
    private val installTimeCache = ConcurrentHashMap<String, Long>()
    private val previousPeriodCache = ConcurrentHashMap<StatsPeriod, Pair<Long, Long>>()
    private val completedDaysCache = ConcurrentHashMap<Long, Long>()

    fun hasUsagePermission(): Boolean {
        val appOps = context.getSystemService(Context.APP_OPS_SERVICE) as AppOpsManager
        val mode = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            appOps.unsafeCheckOpNoThrow(
                AppOpsManager.OPSTR_GET_USAGE_STATS,
                Process.myUid(),
                context.packageName
            )
        } else {
            @Suppress("DEPRECATION")
            appOps.checkOpNoThrow(
                AppOpsManager.OPSTR_GET_USAGE_STATS,
                Process.myUid(),
                context.packageName
            )
        }
        return mode == AppOpsManager.MODE_ALLOWED
    }

    fun clearCache() {
        appNameCache.clear()
        appIconCache.clear()
        systemFlagCache.clear()
        installTimeCache.clear()
        previousPeriodCache.clear()
        completedDaysCache.clear()
    }

    fun getTodayUsageMinutes(): Long {
        if (!hasUsagePermission()) return 0L
        val (start, end) = getPeriodBounds(StatsPeriod.DAY, isPrevious = false)
        val totalMillis = queryUsageTimeMap(start, end).values.sum()
        return totalMillis / 60_000L
    }

    fun getTrackerInstallTime(): Long {
        return getPackageInstallTime(context.packageName)
            .takeIf { it > 0L }
            ?: (System.currentTimeMillis() - 86_400_000L)
    }

    fun getPackageInstallTime(packageName: String): Long {
        installTimeCache[packageName]?.let { return it }
        val installedAt = try {
            val pkgInfo = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                packageManager.getPackageInfo(packageName, PackageManager.PackageInfoFlags.of(0))
            } else {
                @Suppress("DEPRECATION")
                packageManager.getPackageInfo(packageName, 0)
            }
            pkgInfo.firstInstallTime
        } catch (e: Exception) {
            0L
        }
        if (installedAt > 0L) {
            installTimeCache[packageName] = installedAt
        }
        return installedAt
    }

    fun getUsageForPeriod(period: StatsPeriod, hideSystemAndSelf: Boolean = false): List<AppUsageModel> {
        val (start, end) = getPeriodBounds(period, isPrevious = false)
        return getUsageForCustomRange(start, end, hideSystemAndSelf)
    }

    fun getUsageForCustomRange(
        startTime: Long,
        endTime: Long,
        hideSystemAndSelf: Boolean = false,
        perAppInstallTime: Boolean = false
    ): List<AppUsageModel> {
        val safeEnd = max(endTime, startTime + 1000L)
        val usageMap = queryUsageTimeMap(startTime, safeEnd)
        val ownPkg = context.packageName

        return usageMap.entries
            .filter { (pkg, duration) ->
                duration > 1000L && (!hideSystemAndSelf || pkg != ownPkg)
            }
            .mapNotNull { (pkg, rawDuration) ->
                val isSys = isPureSystemPackage(pkg)
                if (hideSystemAndSelf && isSys) return@mapNotNull null

                val name = getAppName(pkg) ?: return@mapNotNull null
                val icon = getAppIcon(pkg)
                val appInstalledAt = getPackageInstallTime(pkg)

                val finalDuration = if (perAppInstallTime && appInstalledAt > startTime && appInstalledAt < safeEnd) {
                    val subMap = queryUsageTimeMap(appInstalledAt, safeEnd)
                    subMap[pkg] ?: rawDuration
                } else {
                    rawDuration
                }

                if (finalDuration <= 1000L) return@mapNotNull null

                AppUsageModel(
                    appName = name,
                    packageName = pkg,
                    timeInMillis = finalDuration,
                    icon = icon,
                    isSystemApp = isSys,
                    installTimeMillis = appInstalledAt
                )
            }
    }

    fun getPreviousPeriodTotalTime(period: StatsPeriod): Long {
        val now = System.currentTimeMillis()
        previousPeriodCache[period]?.let { (timestamp, cachedValue) ->
            if (now - timestamp < 60_000L) return cachedValue
        }
        val (start, end) = getPeriodBounds(period, isPrevious = true)
        val total = queryUsageTimeMap(start, end).values.sum()
        previousPeriodCache[period] = now to total
        return total
    }

    fun getWeeklyDailyBreakdown(): List<Long> {
        val result = MutableList(7) { 0L }
        val now = System.currentTimeMillis()

        val todayStartCal = Calendar.getInstance().apply {
            timeInMillis = now
            set(Calendar.HOUR_OF_DAY, 0)
            set(Calendar.MINUTE, 0)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
        }
        val todayStartMillis = todayStartCal.timeInMillis

        val calendar = Calendar.getInstance().apply {
            firstDayOfWeek = Calendar.MONDAY
            timeInMillis = todayStartMillis
        }

        val currentDayOfWeek = calendar.get(Calendar.DAY_OF_WEEK)
        val diffToMonday = if (currentDayOfWeek == Calendar.SUNDAY) -6 else Calendar.MONDAY - currentDayOfWeek
        calendar.add(Calendar.DAY_OF_MONTH, diffToMonday)

        for (dayIndex in 0 until 7) {
            val dayStart = calendar.timeInMillis
            calendar.add(Calendar.DAY_OF_MONTH, 1)
            val dayEnd = calendar.timeInMillis.coerceAtMost(now)

            if (dayStart < now && dayEnd > dayStart) {
                if (dayEnd <= todayStartMillis) {
                    val cached = completedDaysCache[dayStart]
                    if (cached != null) {
                        result[dayIndex] = cached
                    } else {
                        val computed = queryUsageTimeMap(dayStart, dayEnd).values.sum()
                        completedDaysCache[dayStart] = computed
                        result[dayIndex] = computed
                    }
                } else {
                    result[dayIndex] = queryUsageTimeMap(dayStart, dayEnd).values.sum()
                }
            }
        }
        return result
    }

    private fun getPeriodBounds(period: StatsPeriod, isPrevious: Boolean): Pair<Long, Long> {
        val now = System.currentTimeMillis()
        val cal = Calendar.getInstance()

        return when (period) {
            StatsPeriod.DAY -> {
                cal.timeInMillis = now
                cal.set(Calendar.HOUR_OF_DAY, 0)
                cal.set(Calendar.MINUTE, 0)
                cal.set(Calendar.SECOND, 0)
                cal.set(Calendar.MILLISECOND, 0)
                val todayStart = cal.timeInMillis
                val elapsedToday = max(1L, now - todayStart)

                if (!isPrevious) {
                    todayStart to now
                } else {
                    cal.add(Calendar.DAY_OF_YEAR, -1)
                    val yesterdayStart = cal.timeInMillis
                    yesterdayStart to (yesterdayStart + elapsedToday)
                }
            }

            StatsPeriod.WEEK -> {
                cal.timeInMillis = now
                cal.firstDayOfWeek = Calendar.MONDAY
                cal.set(Calendar.HOUR_OF_DAY, 0)
                cal.set(Calendar.MINUTE, 0)
                cal.set(Calendar.SECOND, 0)
                cal.set(Calendar.MILLISECOND, 0)
                val dayOfWeek = cal.get(Calendar.DAY_OF_WEEK)
                val diffToMonday = if (dayOfWeek == Calendar.SUNDAY) -6 else Calendar.MONDAY - dayOfWeek
                cal.add(Calendar.DAY_OF_MONTH, diffToMonday)
                val weekStart = cal.timeInMillis
                val elapsedWeek = max(1L, now - weekStart)

                if (!isPrevious) {
                    weekStart to now
                } else {
                    cal.add(Calendar.WEEK_OF_YEAR, -1)
                    val prevWeekStart = cal.timeInMillis
                    prevWeekStart to (prevWeekStart + elapsedWeek)
                }
            }

            StatsPeriod.MONTH -> {
                cal.timeInMillis = now
                cal.set(Calendar.DAY_OF_MONTH, 1)
                cal.set(Calendar.HOUR_OF_DAY, 0)
                cal.set(Calendar.MINUTE, 0)
                cal.set(Calendar.SECOND, 0)
                cal.set(Calendar.MILLISECOND, 0)
                val monthStart = cal.timeInMillis
                val elapsedMonth = max(1L, now - monthStart)

                if (!isPrevious) {
                    monthStart to now
                } else {
                    cal.add(Calendar.MONTH, -1)
                    val prevMonthStart = cal.timeInMillis
                    prevMonthStart to (prevMonthStart + elapsedMonth)
                }
            }
        }
    }

    private fun queryUsageTimeMap(startTime: Long, endTime: Long): Map<String, Long> {
        if (!hasUsagePermission() || endTime <= startTime) return emptyMap()

        val eventsTimeMap = HashMap<String, Long>()
        val resumeTimeMap = HashMap<String, Long>()

        val events: UsageEvents? = try {
            usageStatsManager.queryEvents(startTime, endTime)
        } catch (e: Exception) {
            null
        }

        val event = UsageEvents.Event()
        if (events != null) {
            while (events.hasNextEvent()) {
                events.getNextEvent(event)
                val pkg = event.packageName ?: continue
                val timestamp = event.timeStamp

                when (event.eventType) {
                    UsageEvents.Event.ACTIVITY_RESUMED -> {
                        resumeTimeMap[pkg] = timestamp.coerceAtLeast(startTime)
                    }
                    UsageEvents.Event.ACTIVITY_PAUSED,
                    UsageEvents.Event.ACTIVITY_STOPPED -> {
                        val start = resumeTimeMap.remove(pkg)
                        if (start != null && timestamp > start) {
                            val delta = min(timestamp, endTime) - start
                            if (delta > 0) {
                                eventsTimeMap[pkg] = (eventsTimeMap[pkg] ?: 0L) + delta
                            }
                        }
                    }
                }
            }
        }

        for ((pkg, start) in resumeTimeMap) {
            if (endTime > start && (endTime - start) < 2 * 60 * 60 * 1000L) {
                eventsTimeMap[pkg] = (eventsTimeMap[pkg] ?: 0L) + (endTime - start)
            }
        }

        val fiveDaysAgo = System.currentTimeMillis() - 5L * 24L * 3600L * 1000L
        if (eventsTimeMap.isEmpty() || startTime < fiveDaysAgo) {
            val aggregatedMap = HashMap<String, Long>()
            val stats = try {
                usageStatsManager.queryUsageStats(
                    UsageStatsManager.INTERVAL_DAILY,
                    startTime,
                    endTime
                )
            } catch (e: Exception) {
                emptyList()
            }
            stats?.forEach { usageStat ->
                val fgTime = usageStat.totalTimeInForeground
                if (fgTime > 0 && usageStat.lastTimeUsed in startTime..endTime) {
                    val pkg = usageStat.packageName
                    aggregatedMap[pkg] = (aggregatedMap[pkg] ?: 0L) + fgTime
                }
            }

            val maxAllowedDuration = endTime - startTime
            for ((pkg, aggTime) in aggregatedMap) {
                val eventTime = eventsTimeMap[pkg] ?: 0L
                eventsTimeMap[pkg] = max(eventTime, aggTime).coerceAtMost(maxAllowedDuration)
            }
        }

        return eventsTimeMap
    }

    private fun isPureSystemPackage(packageName: String): Boolean {
        systemFlagCache[packageName]?.let { return it }
        val isSys = try {
            val appInfo = packageManager.getApplicationInfo(packageName, 0)
            val isSystemFlag = (appInfo.flags and ApplicationInfo.FLAG_SYSTEM) != 0
            val isUpdatedSystem = (appInfo.flags and ApplicationInfo.FLAG_UPDATED_SYSTEM_APP) != 0
            val hasLaunchIntent = packageManager.getLaunchIntentForPackage(packageName) != null
            (isSystemFlag && !isUpdatedSystem && !hasLaunchIntent) ||
                packageName.contains("launcher", ignoreCase = true) ||
                packageName == "com.android.systemui"
        } catch (e: Exception) {
            false
        }
        systemFlagCache[packageName] = isSys
        return isSys
    }

    private fun getAppName(packageName: String): String? {
        appNameCache[packageName]?.let { return it }
        val resolvedName = try {
            val appInfo = packageManager.getApplicationInfo(packageName, 0)
            packageManager.getApplicationLabel(appInfo).toString()
        } catch (e: PackageManager.NameNotFoundException) {
            packageName.substringAfterLast('.').replaceFirstChar { it.uppercase() }
        }
        appNameCache[packageName] = resolvedName
        return resolvedName
    }

    private fun getAppIcon(packageName: String): Drawable? {
        appIconCache[packageName]?.let { return it }
        return try {
            val icon = packageManager.getApplicationIcon(packageName)
            appIconCache[packageName] = icon
            icon
        } catch (e: Exception) {
            null
        }
    }
}
