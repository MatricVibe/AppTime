package com.example.apptime

import android.content.SharedPreferences
import android.os.Build
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableLongStateOf
import androidx.compose.runtime.mutableStateOf

import androidx.lifecycle.ViewModel
import androidx.lifecycle.ViewModelProvider
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.Dispatchers

import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import kotlin.math.roundToInt

class MainViewModel(
    private val usageHelper: UsageStatsHelper,
    private val repository: FirebaseRepository,
    private val prefs: SharedPreferences
) : ViewModel() {

    val ui = UiSettings(
        prefs = prefs,
        supportsDynamic = Build.VERSION.SDK_INT >= Build.VERSION_CODES.S
    )

    val trackerInstallTimeMillis: Long = run {
        val sysInstall = usageHelper.getTrackerInstallTime()
        val saved = prefs.getLong("tracker_first_install_ms", 0L)
        if (saved <= 0L) {
            prefs.edit().putLong("tracker_first_install_ms", sysInstall).apply()
            sysInstall
        } else {
            minOf(sysInstall, saved)
        }
    }

    var hasPermission by mutableStateOf(usageHelper.hasUsagePermission())
        private set

    var selectedPeriod by mutableStateOf(StatsPeriod.DAY)
        private set

    var usageList by mutableStateOf<List<AppUsageModel>>(emptyList())
        private set

    var todayUsageSnapshot by mutableStateOf<List<AppUsageModel>>(emptyList())
        private set

    var previousTotalMillis by mutableLongStateOf(0L)
        private set

    var weeklyBreakdown by mutableStateOf(List(7) { 0L })
        private set

    var todayMinutes by mutableLongStateOf(0L)
        private set

    var leaderboard by mutableStateOf<List<LeaderboardEntry>>(emptyList())
        private set

    var currentUid by mutableStateOf(repository.currentUid)
        private set

    var isLoading by mutableStateOf(false)
        private set

    var statusMessage by mutableStateOf<String?>(null)
        private set

    var needsNickname by mutableStateOf(false)
        private set
    var isCheckingUsername by mutableStateOf(false)
        private set
    var usernameError by mutableStateOf<String?>(null)
        private set

    // ВОТ ТЕ САМЫЕ ПРОПАВШИЕ ПЕРЕМЕННЫЕ
    val playerName: String
        get() = ui.nickname

    fun updatePlayerName(newName: String) {
        ui.nickname = newName
    }

    init {
        checkDailyLoginStreak()
        refreshLocalUsage()
        loadLeaderboard()
        
        if (repository.getCurrentNickname().isBlank()) {
            needsNickname = true
        }
    }

    private fun checkDailyLoginStreak() {
        val format = SimpleDateFormat("yyyy-MM-dd", Locale.getDefault())
        val todayStr = format.format(Date())
        val yesterdayStr = format.format(Date(System.currentTimeMillis() - 86_400_000L))
        val lastLoginStr = prefs.getString("last_login_date", "").orEmpty()
        
        if (lastLoginStr.isEmpty()) {
            ui.streakDays = 1
            prefs.edit().putString("last_login_date", todayStr).apply()
        } else if (lastLoginStr != todayStr) {
            if (lastLoginStr == yesterdayStr) {
                ui.streakDays += 1
            } else {
                ui.streakDays = 1
            }
            prefs.edit().putString("last_login_date", todayStr).apply()
        }
    }

    fun registerUsername(newUsername: String) {
        val cleanInput = newUsername.trim()
        
        if (cleanInput.length < 3) {
            usernameError = "Минимум 3 символа"
            return
        }
        if (!cleanInput.matches(Regex("^[a-zA-Z0-9_]+$"))) {
            usernameError = "Только буквы, цифры и _"
            return
        }

        viewModelScope.launch {
            isCheckingUsername = true
            usernameError = null
            try {
                val isAvailable = repository.isUsernameAvailable(cleanInput)
                
                if (isAvailable) {
                    val latestMinutes = withContext(Dispatchers.IO) {
                        usageHelper.getTodayUsageMinutes()
                    }
                    todayMinutes = latestMinutes
                    
                    ui.nickname = cleanInput 
                    ui.username = cleanInput 
                    
                    repository.registerUsernameAndSubmit(ui.nickname, ui.username, latestMinutes)
                    
                    needsNickname = false
                    loadLeaderboard()
                    statusMessage = "Ник @$cleanInput успешно занят!"
                } else {
                    usernameError = "Этот юзернейм уже занят"
                }
            } catch (e: Exception) {
                usernameError = "Ошибка сети: ${e.localizedMessage}"
            } finally {
                isCheckingUsername = false
            }
        }
    }

    fun selectPeriod(period: StatsPeriod) {
        selectedPeriod = period
        ui.periodsMask = ui.periodsMask or period.maskBit
        refreshLocalUsage()
    }

    fun setAppLimit(packageName: String, limitMinutes: Int) {
        val updated = ui.appLimitsMap.toMutableMap()
        if (limitMinutes <= 0) updated.remove(packageName) else updated[packageName] = limitMinutes
        ui.appLimitsMap = updated
    }

    fun refreshLocalUsage() {
        hasPermission = usageHelper.hasUsagePermission()
        if (!hasPermission) {
            usageList = emptyList()
            todayUsageSnapshot = emptyList()
            todayMinutes = 0L
            return
        }
        val period = selectedPeriod
        val hideSys = ui.hideSystemApps
        val needChart = period == StatsPeriod.WEEK && ui.showWeeklyChart

        viewModelScope.launch {
            val (periodData, prevTotal, weekBars, dayData) = withContext(Dispatchers.IO) {
                val cur = usageHelper.getUsageForPeriod(period, hideSys)
                val prev = usageHelper.getPreviousPeriodTotalTime(period)
                val bars = if (needChart) usageHelper.getWeeklyDailyBreakdown() else emptyList()
                val day = if (period == StatsPeriod.DAY) cur else usageHelper.getUsageForPeriod(StatsPeriod.DAY, hideSys)
                Quadruple(cur, prev, bars, day)
            }
            usageList = periodData
            previousTotalMillis = prevTotal
            if (weekBars.isNotEmpty()) weeklyBreakdown = weekBars
            todayUsageSnapshot = dayData
            todayMinutes = dayData.sumOf { it.timeInMillis } / 60_000L
        }
    }

    suspend fun loadCustomRangeUsage(
        startTime: Long,
        endTime: Long,
        perAppInstall: Boolean
    ): List<AppUsageModel> = withContext(Dispatchers.IO) {
        if (!usageHelper.hasUsagePermission()) return@withContext emptyList()
        usageHelper.getUsageForCustomRange(
            startTime = startTime,
            endTime = endTime,
            hideSystemAndSelf = ui.hideSystemApps,
            perAppInstallTime = perAppInstall
        )
    }

    fun clearAllCaches() {
        usageHelper.clearCache()
        refreshLocalUsage()
    }

    fun loadLeaderboard() {
        viewModelScope.launch {
            isLoading = true
            statusMessage = null
            try {
                leaderboard = repository.getTop50Leaderboard()
                currentUid = repository.currentUid
            } catch (e: Exception) {
                statusMessage = "Ошибка загрузки: ${e.localizedMessage ?: "неизвестная ошибка"}"
            } finally {
                isLoading = false
            }
        }
    }

    fun submitCurrentResult() {
        viewModelScope.launch {
            isLoading = true
            statusMessage = null
            try {
                val latestMinutes = withContext(Dispatchers.IO) {
                    usageHelper.getTodayUsageMinutes()
                }
                todayMinutes = latestMinutes
                
                repository.submitScore(
                    nickname = ui.nickname, 
                    username = ui.username.removePrefix("@"), 
                    minutes = latestMinutes
                )
                
                currentUid = repository.currentUid
                leaderboard = repository.getTop50Leaderboard()
                statusMessage = "Результат ($latestMinutes мин) отправлен в Топ-50!"
            } catch (e: Exception) {
                statusMessage = "Ошибка отправки: ${e.localizedMessage ?: "неизвестная ошибка"}"
            } finally {
                isLoading = false
            }
        }
    }

    class Factory(
        private val usageHelper: UsageStatsHelper,
        private val repository: FirebaseRepository,
        private val prefs: SharedPreferences
    ) : ViewModelProvider.Factory {
        @Suppress("UNCHECKED_CAST")
        override fun <T : ViewModel> create(modelClass: Class<T>): T {
            return MainViewModel(usageHelper, repository, prefs) as T
        }
    }
}

private data class Quadruple<A, B, C, D>(
    val first: A,
    val second: B,
    val third: C,
    val fourth: D
)
